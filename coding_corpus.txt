# Production-ready GitHub Open Issues Scraper - Full Working Example
# Includes pagination, rate limiting, exponential backoff, retries, error handling, JSON export

import requests
import time
import json
from typing import List, Dict, Any, Optional
from requests.exceptions import RequestException, HTTPError

class GitHubIssueScraper:
    """Fully robust GitHub Issues scraper with rate limiting and retries."""
    
    def __init__(self, token: Optional[str] = None):
        self.token = token
        self.headers = {
            "Accept": "application/vnd.github.v3+json",
            "User-Agent": "HRM-Coding-Assistant"
        }
        if token:
            self.headers["Authorization"] = f"token {token}"
        self.session = requests.Session()

    def _handle_rate_limit(self, response):
        remaining = int(response.headers.get("X-RateLimit-Remaining", 60))
        if remaining < 5:
            reset_time = int(response.headers.get("X-RateLimit-Reset", time.time() + 60))
            sleep_time = max(0, reset_time - time.time() + 2)
            print(f"Rate limit near zero — sleeping {sleep_time:.0f}s")
            time.sleep(sleep_time)

    def get_all_open_issues(self, owner: str, repo: str, max_pages: int = 50) -> List[Dict]:
        """Main method: returns list of all open issues + saves to JSON."""
        url = f"https://api.github.com/repos/{owner}/{repo}/issues"
        issues = []
        page = 1

        while page <= max_pages:
            params = {"state": "open", "per_page": 100, "page": page, "sort": "created", "direction": "asc"}
            try:
                resp = self.session.get(url, headers=self.headers, params=params, timeout=15)
                resp.raise_for_status()
                
                self._handle_rate_limit(resp)
                
                batch = resp.json()
                if not batch:
                    break
                
                issues.extend(batch)
                print(f"Page {page}: {len(batch)} issues fetched")
                
                if len(batch) < 100:
                    break  # last page
                
                page += 1
            except HTTPError as e:
                if e.response.status_code == 429:
                    self._handle_rate_limit(e.response)
                    continue
                print(f"HTTP error on page {page}: {e}")
                time.sleep(10)
            except RequestException as e:
                print(f"Network error on page {page}: {e}")
                time.sleep(5)
        
        # Save to JSON
        filename = f"{owner}_{repo}_open_issues.json"
        with open(filename, "w", encoding="utf-8") as f:
            json.dump(issues, f, indent=2, ensure_ascii=False)
        
        print(f"Saved {len(issues)} issues to {filename}")
        return issues
